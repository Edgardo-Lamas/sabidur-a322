### Tarea 3: Componente Card unificado

**Files:**
- Create: `src/components/ui/Card.jsx`

El sitio tiene `.recommended-card` en CSS y cards ad-hoc en artículos. Crear un componente reutilizable.

**Paso 1: Crear `src/components/ui/Card.jsx`**

```jsx
import React from 'react';
import { Link } from 'react-router-dom';

/**
 * Card — Componente de tarjeta unificado
 *
 * @param {'default'|'dark'|'light'} variant
 * @param {string} to - ruta interna (React Router)
 * @param {string} href - enlace externo
 * @param {string} category - etiqueta superior (gold, uppercase)
 * @param {string} title
 * @param {string} description
 * @param {string} image - URL de imagen
 * @param {string} className - clases adicionales
 */
const Card = ({
  variant = 'default',
  to,
  href,
  category,
  title,
  description,
  image,
  className = '',
  children,
  ...props
}) => {
  const variantClasses = {
    default: 'bg-white border border-sabiduria-gray/10 text-sabiduria-navy',
    dark: 'bg-sabiduria-navy text-white hover:bg-[var(--color-navy-hover)]',
    light: 'bg-sabiduria-bg border border-sabiduria-gray/10 text-sabiduria-navy',
  };

  const baseClasses = `block rounded-sm overflow-hidden shadow-sm transition-all duration-200 hover:shadow-md hover:-translate-y-0.5 ${variantClasses[variant]} ${className}`;

  const inner = (
    <>
      {image && (
        <img
          src={image}
          alt={title || ''}
          className="w-full h-48 object-cover"
          loading="lazy"
        />
      )}
      <div className="p-5">
        {category && (
          <span className="text-xs font-bold uppercase tracking-[0.1em] text-sabiduria-gold block mb-2">
            {category}
          </span>
        )}
        {title && (
          <h3 className="font-heading font-semibold text-base leading-snug mb-2">
            {title}
          </h3>
        )}
        {description && (
          <p className="text-sm text-sabiduria-gray leading-relaxed">{description}</p>
        )}
        {children}
      </div>
    </>
  );

  if (to) return <Link to={to} className={baseClasses} {...props}>{inner}</Link>;
  if (href) return <a href={href} className={baseClasses} target="_blank" rel="noopener noreferrer" {...props}>{inner}</a>;
  return <div className={baseClasses} {...props}>{inner}</div>;
};

export default Card;
```

```jsx
import React from 'react';
import { Link } from 'react-router-dom';
import SEO from '../components/SEO';

// Importar todos los JSON de knowledge dinámicamente
const knowledgeModules = import.meta.glob('../data/knowledge/*.json', { eager: true });

const TeologiaSistematica = () => {
  const topics = Object.entries(knowledgeModules).map(([path, module]) => {
    const fileName = path.split('/').pop().replace('.json', '');
    return {
      slug: fileName,
      title: module.default?.title || module.title || fileName,
      description: module.default?.description || module.description || '',
    };
  });

  return (
    <>
      <SEO
        title="Teología Sistemática"
        description="Base de conocimiento teológico reformado: bibliología, cristología, soteriología, escatología y más."
      />
      <div className="max-w-4xl mx-auto px-4 py-16">
        <h1 className="text-4xl font-serif text-sabiduria-navy mb-4">Teología Sistemática</h1>
        <p className="text-sabiduria-gray mb-12">
          Fundamentos doctrinales de la fe reformada, organizados por disciplina teológica.
        </p>
        <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
          {topics.map((topic) => (
            <div key={topic.slug} className="border border-sabiduria-gray/20 rounded-sm p-5 hover:border-sabiduria-gold transition-colors">
              <h2 className="font-heading font-semibold text-lg text-sabiduria-navy mb-1">{topic.title}</h2>
              {topic.description && <p className="text-sm text-sabiduria-gray">{topic.description}</p>}
            </div>
          ))}
        </div>
      </div>
    </>
  );
};

export default TeologiaSistematica;
```
